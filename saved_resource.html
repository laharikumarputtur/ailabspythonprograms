Program 12: 8 Puzzle Using Different Heuristics

Aim: To solve the 8 Puzzle using A* search with misplaced-tile and Manhattan-distance heuristics.

Program

program_12_eight_puzzle_heuristics.py
import heapq

start = (1, 2, 3, 4, 0, 6, 7, 5, 8)
goal = (1, 2, 3, 4, 5, 6, 7, 8, 0)

def misplaced(state):
    return sum(tile != 0 and tile != goal[i] for i, tile in enumerate(state))

def manhattan(state):
    total = 0
    for i, tile in enumerate(state):
        if tile:
            j = goal.index(tile)
            total += abs(i // 3 - j // 3) + abs(i % 3 - j % 3)
    return total

def solve(heuristic):
    queue = [(heuristic(start), 0, start)]
    best = {start: 0}
    while queue:
        _, steps, state = heapq.heappop(queue)
        if steps != best[state]:
            continue
        if state == goal:
            return steps
        zero = state.index(0)
        for move in (-3, 3, -1, 1):
            new = zero + move
            if 0 <= new < 9 and not (move == -1 and zero % 3 == 0) \
                    and not (move == 1 and zero % 3 == 2):
                board = list(state)
                board[zero], board[new] = board[new], board[zero]
                child = tuple(board)
                if steps + 1 < best.get(child, 99):
                    best[child] = steps + 1
                    heapq.heappush(queue, (steps + 1 + heuristic(child),
                                           steps + 1, child))
    return -1

print("Moves using Misplaced Tiles:", solve(misplaced))
print("Moves using Manhattan Distance:", solve(manhattan))
